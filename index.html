let dashboardData = null; async function fetchDashboard() { try { // Make sure to use your live HTTPS Render URL for phone access const res = await fetch('https://mohit-mangment-system.onrender.com/api/dashboard-summary'); dashboardData = await res.json(); if (dashboardData.error) { console.error('API Error:', dashboardData.error); return; } // Initialize extra attended and bunked counts for each course if (dashboardData.courses) { dashboardData.courses.forEach(course => { course.attendedCount = 0; course.bunkedCount = 0; }); } // Populate Profile Data document.getElementById('s-name').textContent = dashboardData.student?.name || 'N/A'; document.getElementById('s-email').textContent = dashboardData.student?.email || 'N/A'; document.getElementById('s-enroll').textContent = dashboardData.student?.enrollment || 'N/A'; renderDashboard(); } catch (err) { console.error('Failed to load dashboard:', err); } } // Handler to attend an extra class (+1 present, +1 total) function attendCourse(courseId) { const course = dashboardData.courses.find(c => c.course_id === courseId); if (course) { course.attendedCount = (course.attendedCount || 0) + 1; renderDashboard(); } } // Handler to bunk an extra class (+0 present, +1 total) function bunkCourse(courseId) { const course = dashboardData.courses.find(c => c.course_id === courseId); if (course) { course.bunkedCount = (course.bunkedCount || 0) + 1; renderDashboard(); } } function renderDashboard() { if (!dashboardData) return; const courses = dashboardData.courses || []; let updatedTotalClasses = 0; let updatedPresentClasses = 0; const coursesContainer = document.getElementById('courses-container'); coursesContainer.innerHTML = ''; courses.forEach(course => { const originalPresent = parseInt(course.present_classes || 0); const originalTotal = parseInt(course.total_classes || 0); const extraAttended = course.attendedCount || 0; const extraBunked = course.bunkedCount || 0; // Calculate simulated attendance const newPresent = originalPresent + extraAttended; const newTotal = originalTotal + extraAttended + extraBunked; const newPercentage = newTotal > 0 ? (newPresent / newTotal) * 100 : 0; const isSafe = newPercentage >= 75; // Accumulate overall aggregates updatedPresentClasses += newPresent; updatedTotalClasses += newTotal; const card = document.createElement('div'); card.className = 'course-card'; card.innerHTML = `
${course.course_name}
${course.course_code} • ${course.course_type.toUpperCase()}
👨‍🏫 ${course.faculty_name}
${newPercentage.toFixed(1)}%
Attended: ${newPresent}/${newTotal} Sem: ${course.semester}
`; coursesContainer.appendChild(card); }); // Recalculate Top Overview Card const overallPercentage = updatedTotalClasses > 0 ? ((updatedPresentClasses / updatedTotalClasses) * 100).toFixed(2) : 0; document.getElementById('s-pct').textContent = `${overallPercentage}%`; document.getElementById('s-present').textContent = updatedPresentClasses; document.getElementById('s-total').textContent = updatedTotalClasses; } fetchDashboard();
